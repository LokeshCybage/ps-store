1) Ensure required Secrets exist (no secrets are stored in values.yaml):

  - Postgres credentials secret (default name: {{ .Values.postgres.credentialsSecret.name }})
      keys:
        - {{ .Values.postgres.credentialsSecret.userKey }}
        - {{ .Values.postgres.credentialsSecret.passwordKey }}

  - Game catalog secret (default name: {{ .Values.gameCatalogService.existingSecret.name }})
      keys:
        - {{ .Values.gameCatalogService.existingSecret.databaseUrlKey }}
        - {{ .Values.gameCatalogService.existingSecret.jwtSecretKey }}

  - User service secret (default name: {{ .Values.userService.existingSecret.name }})
      keys:
        - {{ .Values.userService.existingSecret.connectionStringKey }}
        - {{ .Values.userService.existingSecret.jwtSecretKey }}

  - Order service secret (default name: {{ .Values.orderService.existingSecret.name }})
      keys:
        - {{ .Values.orderService.existingSecret.databaseUrlKey }}
        - {{ .Values.orderService.existingSecret.jwtSecretKey }}

2) Install:

  helm install {{ .Release.Name }} {{ .Chart.Name }} -n {{ .Release.Namespace }} --create-namespace

3) Render (debug):

  helm template {{ .Release.Name }} {{ .Chart.Name }} -n {{ .Release.Namespace }} --debug
  helm template {{ .Release.Name }} {{ .Chart.Name }} -n {{ .Release.Namespace }} -f values-prod.yaml --debug
